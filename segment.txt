    const sd=kind==='right'?1:-1, arms=[[sd>0?P.sR:P.sL,sd>0?'eR':'eL',sd]];
    if(four&&id==='four_arms') arms.push([sd>0?P.s2L:P.s2R,sd>0?'e2L':'e2R',-sd]); // the diagonal lower arm hooks in sync
    // free arm holds a guard beside the chin
    const oS=sd>0?P.sL:P.sR, oe=sd>0?'eL':'eR'; P[oe]=guard(oS,-sd,P[oe],env);
    arms.forEach(([S,ek,sg],i)=>{
      const hookY=i?1.0:0;
      guard(S,sg,0,env);
      S.x = _lerp(.62,.05,coil)*env + 1.5*ext*amp*(i?.9:1) + (1-env)*S.x;   // cocked back beside the ribs -> horizontal at contact
      S.y = sg*(.42*env*(1-ext) + .14*coil - (.5+hookY*.3)*ext);           // in across the body, then out to compensate the torso twist
      S.z = sg*(.06+.08*hookY)*ext;
      P[ek] = _lerp(_lerp(1.85,2.35,coil),.06,ext);
    });
    P.chY += sd*(-.30*coil + .60*ext)*amp; P.hipsRY += sd*(-.14*coil + .32*ext)*amp;
    P.chX += (.10*coil - .24*ext)*amp; P.hdY += -sd*.16*ext*amp; P.hdX += .08*ext;
    const lead=sd>0?'L':'R', rear=sd>0?'R':'L';
    P['t'+lead+'x']=_lerp(P['t'+lead+'x'],.55,env); P['k'+lead]=_lerp(P['k'+lead],.5,env);
    P['t'+rear+'x']=_lerp(P['t'+rear+'x'],-.5*(cb?.5:1),env); P['k'+rear]=_lerp(P['k'+rear],.95,env); P['f'+rear]=_lerp(P['f'+rear],.75,env);
    P.hipsZ += -.16*ext*amp*(xl?1.4:1); P.hipsY += -.07*env-.04*ext;
  } else if(kind==='heavy'||kind==='slam'){
    // Two-handed finisher: rear back with both fists high, then throw the whole body forward and slam down.
    const arms=[[P.sL,'eL',-1,0],[P.sR,'eR',1,0]]; if(four) arms.push([P.s2L,'e2L',-1,1],[P.s2R,'e2R',1,1]);
    arms.forEach(([S,ek,sg,low])=>{
      S.x = ( _lerp(.5,low?.9:2.35,coil) + 0 )*(1-ext) + ext*(low?1.5:1.45)*amp;   // overhead (upper) / cocked (lower) -> slam forward
      S.y = sg*(.3*(1-ext)); S.z = sg*.1*ext;
      P[ek] = _lerp(_lerp(1.6,low?2.2:1.1,coil),.12,ext);
    });
    P.chX += (.5*coil - .8*ext)*amp; P.hipsRX += (.1*coil-.22*ext)*amp; P.hdX += (-.25*coil+.45*ext);
    P.chY += (at.side?1:-1)*.2*ext; P.hipsY += -.15*env-.15*ext*(cb?0:1); P.hipsZ += -.3*ext*amp-.05*coil;
    P.tLx=_lerp(P.tLx,.75,env); P.kL=_lerp(P.kL,.9,env); P.tRx=_lerp(P.tRx,-.4,env); P.kR=_lerp(P.kR,1.2,env); P.fR=_lerp(P.fR,.8,env);
    if(cb){ P.chX+=-.4*ext; P.hipsY+=.08*ext; }
  } else if(kind==='shoot'){
    const k=1-_sm(t/.55); P.sR.x+=-.35*k; P.eR+=.5*k; P.chX+=.1*k; P.hdX+=-.08*k;
  }
}
/* Main animator. inp: {pos, fwd:{x,z}, grounded, vy, flying, hover, swimming, carrying, superT, atk:{kind,t,side,id,amp,tc},
   aim, flee, hurt(0..1), dead(0..1), hp, id, powerScale} */
function animateRig(parts,dt,inp){
  if(!parts||!parts.hips||dt<=0) return;
  if(parts.fx) tickFx(parts,dt);
  const A=parts.anim, P=_P; resetPose(P);
  A.t+=dt;
  // world speed from position delta (works identically for player, replicas, NPCs, enemies; teleports are ignored)
  let spd=0, fwdV=1, latV=0;
  if(inp.pos){ if(A.px!==null){ const dx=inp.pos.x-A.px, dz=inp.pos.z-A.pz, d=Math.hypot(dx,dz); spd=d/dt; if(spd>90){ spd=0; } else if(inp.fwd&&d>1e-4){ fwdV=(dx*inp.fwd.x+dz*inp.fwd.z)/d; latV=(dx*-inp.fwd.z+dz*inp.fwd.x)/d; } } A.px=inp.pos.x; A.pz=inp.pos.z; }
  A.spd=_damp(A.spd,spd,14,dt);
  const sc=Math.max(.4,(parts.dims.total*(inp.ws||PS))/(2.7*PS)); // stride scales with the body's world size
  const s=Math.min(4.2,A.spd/(3.6*sc)); // normalised: 1 = walk, ~2 = run, 3+ = sprint
  const grounded=inp.grounded!==false;
  const r=_sm((s-1.25)/1.1), q=_sm((s-2.5)/1.2), mv=_sm(s/.35);
  A.vyS=_damp(A.vyS,inp.vy||0,12,dt);
  // landing detection
  if(!A.wasGround && grounded){ A.land=Math.min(1,.35+Math.max(0,-(A.lastVy||0))/22); }
  A.wasGround=grounded; A.lastVy=inp.vy||0; A.land=Math.max(0,A.land-dt*3.4);
  const dir=(fwdV<-.15&&inp.fwd)?-1:1; // backpedal
  A.ph += dt*(5.2+2.5*Math.min(s,3.4)+1.2*q)*dir*mv; 
  const ph=A.ph, sL=Math.sin(ph), sR=-sL, cL=Math.cos(ph), cR=-cL;
  const dead=inp.dead||0, hurt=inp.hurt||0, chg=inp.charge||0;
  // ---- locomotion ----
  if(grounded && !inp.flying && !inp.swimming){
    const legAmp=_lerp(_lerp(.5,.95,r),1.15,q), kmax=_lerp(.7,1.75,r), lean=-( .04+.16*r+.22*q );
    P.tLx = sL*legAmp*mv; P.tRx = sR*legAmp*mv;
    const kf=(c,sn)=> Math.max(0,c*.8+.2*sn)*kmax + (1-r)*.12 + r*.18*Math.max(0,-c);      // flex on the forward swing, slight flex at contact
    P.kL = kf(cL,sL)*mv; P.kR = kf(cR,sR)*mv;
    P.fL = -(P.tLx-P.kL)*.55 + Math.max(0,-cL)*.35*mv; P.fR = -(P.tRx-P.kR)*.55 + Math.max(0,-cR)*.35*mv;
    P.hipsY += (Math.cos(ph*2)*(.025+.05*r+.05*q) - (.02+.07*r+.08*q))*mv;
    P.hipsRY += sL*.20*mv*(1-.4*r); P.chY += -sL*.30*mv*(1-.2*r)*(1+.4*q); P.hipsRZ += sL*.045*mv*(1-r); P.chZ += -sL*.03*mv;
    P.chX += lean*mv; P.hdX += -lean*.75*mv; P.hdY += sL*.1*mv;
    P.hipsZ += latV*.0; 
    // arms swing opposite the legs, elbows fold more as speed rises
    const aa=_lerp(_lerp(.55,1.0,r),1.25,q);
    P.sL.x = -sL*aa*mv; P.sR.x = -sR*aa*mv;
    P.eL = (_lerp(.2,1.45,r)+.28*Math.max(0,-sL)*(.4+r))*mv + .12*(1-mv); P.eR = (_lerp(.2,1.45,r)+.28*Math.max(0,-sR)*(.4+r))*mv + .12*(1-mv);
    P.sL.z = -.05-.08*r; P.sR.z = .05+.08*r; P.sL.y=r*.12; P.sR.y=-r*.12;
    if(parts.leftArm2){ P.s2L.x=-sR*aa*.8*mv; P.s2R.x=-sL*aa*.8*mv; P.e2L=.35+.5*r; P.e2R=.35+.5*r; P.s2L.z=-.12; P.s2R.z=.12; }
    // idle: breathing, weight shift, slow head scan, relaxed arms
    const idle=1-mv, bt=A.t;
    P.chX += Math.sin(bt*1.9)*.02*idle; P.sL.x += (.06+Math.sin(bt*1.9+.6)*.03)*idle; P.sR.x += (.06+Math.sin(bt*1.9+1.2)*.03)*idle;
    P.hipsRZ += Math.sin(bt*.7)*.03*idle; P.chZ += -Math.sin(bt*.7)*.02*idle; P.hdY += Math.sin(bt*.45)*.22*idle; P.hipsY += Math.sin(bt*1.9)*.008*idle;
    P.tLx += .03*idle; P.kL += .04*idle; P.tRx += -.03*idle; P.kR += .07*idle;
    if(parts.leftArm2){ P.s2L.x+=.1*idle; P.s2R.x+=.1*idle; P.e2L+=.5*idle; P.e2R+=.5*idle; }
    // landing squash
    if(A.land>0){ const k=_sm(A.land); P.hipsY += -.2*k; P.tLx+=.5*k; P.tRx+=.35*k; P.kL+=1.15*k; P.kR+=1.0*k; P.chX+=-.3*k; P.hdX+=.15*k; P.sL.x+=.5*k; P.sR.x+=.5*k; P.sL.z+=-.2*k; P.sR.z+=.2*k; }
  } else if(!inp.flying && !inp.swimming){
    // airborne: rising = knees tucked / fists up, falling = legs reach down, arms spread for balance
    const rise=_sm(A.vyS/6), fall=_sm(-A.vyS/8);
    P.tLx=.55*rise+.25*fall; P.tRx=-.4*rise+.05*fall; P.kL=1.25*rise+.35*fall; P.kR=1.35*rise+.5*fall; P.fL=.3; P.fR=.3;
    P.sL.x=-.4+1.0*rise-.2*fall; P.sR.x=-.4+1.0*rise-.2*fall; P.sL.z=-.35-.55*fall; P.sR.z=.35+.55*fall; P.eL=.5+.6*rise; P.eR=.5+.6*rise;
    P.chX=-.12*rise+.1*fall; P.hdX=.1; P.hipsY=.02; P.hipsRZ=0;
    if(parts.leftArm2){ P.s2L.x=.5*rise; P.s2R.x=.5*rise; P.s2L.z=-.5; P.s2R.z=.5; P.e2L=.6; P.e2R=.6; }
  }
  if(inp.climb){ const cph=A.climbPh=(A.climbPh||0)+dt*(inp.climbSpd||0)*3.4, c=Math.sin(cph);
    P.sL.x=2.3+c*.55; P.sR.x=2.3-c*.55; P.eL=.9-c*.55; P.eR=.9+c*.55; P.sL.z=-.1; P.sR.z=.1; P.tLx=.55+c*.6; P.tRx=.55-c*.6; P.kL=.9-c*.5; P.kR=.9+c*.5; P.chX=-.1; P.hdX=-.15; P.hipsY=-.02; if(parts.leftArm2){ P.s2L.x=2.0-c*.4; P.s2R.x=2.0+c*.4; P.e2L=.8; P.e2R=.8; } }
  // ---- flying / hovering / swimming ----
  if(inp.flying){
    if(inp.hover){ // levitate upright, arms low and relaxed, gentle bob
      P.hipsY=Math.sin(A.t*2.2)*.05; P.tLx=.15; P.tRx=-.1; P.kL=.35; P.kR=.55; P.sL.x=.25; P.sR.x=.25; P.sL.z=-.5; P.sR.z=.5; P.eL=.5; P.eR=.5; P.chX=.03; P.fL=.6; P.fR=.6;
      if(parts.leftArm2){ P.s2L.x=.3; P.s2R.x=.3; P.s2L.z=-.6; P.s2R.z=.6; P.e2L=.5; P.e2R=.5; }
    } else { // Superman: arms locked forward over the head, legs streamed behind, slight wobble
      const up=inp.moving?2.85:.55; P.sL.x=up; P.sR.x=up; P.eL=.1; P.eR=.1; P.sL.z=-.08; P.sR.z=.08;
      P.tLx=.12+Math.sin(A.t*9)*.14; P.tRx=.12-Math.sin(A.t*9)*.14; P.kL=.25; P.kR=.25; P.fL=.6; P.fR=.6; P.chX=inp.moving?-.05:.03; P.hdX=inp.moving?.35:0;
      P.hipsRZ=Math.sin(A.t*3)*.03;
    }
  } else if(inp.swimming){
    const sw=A.t*6; P.sL.x=1.4+Math.sin(sw)*1.4; P.sR.x=1.4-Math.sin(sw)*1.4; P.eL=.5+Math.max(0,Math.sin(sw))*.6; P.eR=.5+Math.max(0,-Math.sin(sw))*.6;
    P.tLx=Math.sin(sw*2)*.45; P.tRx=-Math.sin(sw*2)*.45; P.kL=.3+Math.max(0,Math.sin(sw*2))*.4; P.kR=.3+Math.max(0,-Math.sin(sw*2))*.4; P.chY=Math.sin(sw)*.25; P.hdX=.5; 
  }
  // ---- special stances ----
  if(inp.superT>.2 && grounded && !inp.flying){ // XLR8 sprint: arms swept back, deep forward lean, driving knees
    const k=_ez(inp.superT); P.sL.x=_lerp(P.sL.x,-1.5,k); P.sR.x=_lerp(P.sR.x,-1.5,k); P.eL=_lerp(P.eL,.35,k); P.eR=_lerp(P.eR,.35,k); P.chX+=-.35*k; P.hdX+=.3*k; P.sL.z=-.12; P.sR.z=.12;
    P.tLx*=1.25; P.tRx*=1.25; P.kL*=1.2; P.kR*=1.2;
  }
  if(inp.carrying){ for(const [S,ek] of [[P.sL,'eL'],[P.sR,'eR'],[P.s2L,'e2L'],[P.s2R,'e2R']]){ S.x=2.9; S.z*=.3; P[ek]=.25; } P.chX+=.1; P.hdX+=.1; }
  if(inp.flee){ P.sL.x=2.5+Math.sin(A.t*14)*.35; P.sR.x=2.5-Math.sin(A.t*14)*.35; P.eL=.8; P.eR=.8; P.chX+=-.12; P.hdX+=.25; }
  if(inp.aim){ // pistol/blaster: right arm straight out, left hand supports, head turned down the sight line
    P.sR.x=1.55; P.sR.y=-.05; P.eR=.05; P.sL.x=1.3; P.sL.y=-.45; P.eL=.85; P.sR.z=.03; P.chY+=.18; P.chX+=-.05; P.hdY+=-.15;
    P.tLx=_lerp(P.tLx,.25,.6); P.tRx=_lerp(P.tRx,-.25,.6); P.kL+=.15; P.kR+=.15; }
  // ---- melee overlay ----
  if(chg>0.01) chargePose(P,chg,parts);
  if(inp.atk) attackPose(P,parts,inp.atk,dt);
  // ---- hit reaction ----
  if(hurt>0){ const h=_sm(hurt); P.chX+=.42*h; P.hdX+=.4*h; P.hipsRX+=.1*h; P.hipsZ+=.12*h; P.sL.x+=-.6*h; P.sR.x+=-.6*h; P.sL.z+=-.5*h; P.sR.z+=.5*h; P.eL+=.5*h; P.eR+=.5*h; P.tLx+=.35*h; P.kL+=.4*h; P.kR+=.5*h; P.hipsY+=-.08*h; P.chY+=Math.sin(A.t*40)*.05*h; }
  // ---- death: crumple; the whole-body fall is applied to the model root by the caller ----
  if(dead>0){ const d=_sm(dead); P.sL.x=_lerp(P.sL.x,-.5,d); P.sR.x=_lerp(P.sR.x,.2,d); P.sL.z=_lerp(P.sL.z,-.6,d); P.sR.z=_lerp(P.sR.z,.8,d); P.eL=_lerp(P.eL,.5,d); P.eR=_lerp(P.eR,.9,d);
    P.tLx=_lerp(P.tLx,.4,d); P.tRx=_lerp(P.tRx,-.2,d); P.kL=_lerp(P.kL,.9,d); P.kR=_lerp(P.kR,.3,d); P.hdX=_lerp(P.hdX,.35,d); P.hdY=_lerp(P.hdY,.4,d); P.chX=_lerp(P.chX,.15,d); }
  // ---- apply with per-joint smoothing (attacks & hits use a stiff rate so they stay snappy) ----
  const fast=(inp.atk||hurt>0||dead>0)?38:(inp.flying||inp.swimming?12:(grounded?26:14)), rt=fast;
  const ap=(o,k,v)=>{ o.rotation[k]=_damp(o.rotation[k],v,rt,dt); };
  const hb=parts.hipY, hy=hb+P.hipsY; parts.hips.position.y=_damp(parts.hips.position.y,hy,rt,dt); parts.hips.position.z=_damp(parts.hips.position.z,P.hipsZ,rt,dt);
  ap(parts.hips,'x',P.hipsRX); ap(parts.hips,'y',P.hipsRY); ap(parts.hips,'z',P.hipsRZ);
  ap(parts.chest,'x',P.chX); ap(parts.chest,'y',P.chY); ap(parts.chest,'z',P.chZ);
  ap(parts.head,'x',P.hdX); ap(parts.head,'y',P.hdY); ap(parts.head,'z',P.hdZ);
  const J=[[parts.leftArm,P.sL],[parts.rightArm,P.sR],[parts.leftArm2,P.s2L],[parts.rightArm2,P.s2R]];
  for(const [o,S] of J){ if(!o) continue; ap(o,'x',S.x); ap(o,'y',S.y); ap(o,'z',S.z); }
  ap(parts.elbowL,'x',P.eL); ap(parts.elbowR,'x',P.eR); if(parts.elbow2L){ ap(parts.elbow2L,'x',P.e2L); ap(parts.elbow2R,'x',P.e2R); }
  ap(parts.leftLeg,'x',P.tLx); ap(parts.rightLeg,'x',P.tRx); ap(parts.leftLeg,'z',P.tLz); ap(parts.rightLeg,'z',P.tRz);
  ap(parts.kneeL,'x',-P.kL); ap(parts.kneeR,'x',-P.kR); ap(parts.footL,'x',P.fL); ap(parts.footR,'x',P.fR);
  poseImportedAlien(parts,dt,inp,P);
}
// Helper: turn a timer-driven attack ({kind,dur,t}) into the attack input for animateRig.
function atkInput(kind,left,dur,extra){ if(!(left>0)||!kind) return null; const o={kind,t:1-Math.max(0,Math.min(1,left/dur))}; if(extra) Object.assign(o,extra); return o; }

    bx(head,hs*1.16,hs*.76,hs*1.12,yel,0,hs*.6,0); bx(head,hs*.9,.18,.03,blk,0,hs*.56,-hs*.56-.02); bx(head,hs*1.2,.1,hs*1.16,blk,0,hs*.28,0);

    shell.position.y=shellCenterY; shell.scale.setScalar(0.001); g.add(shell); parts.shell=shell; parts.shellBaseY=shellCenterY;
  }
  const o=new THREE.Group(); g.position.y=-0.9; o.add(g); o.rotation.order='YXZ';
  o.userData.parts=parts; o.userData.kind=kind; o.userData.proceduralRoot=g;
  if(IMPORTED_ALIEN_ASSETS[kind]) attachImportedAlienVisual(o,kind);
  return o;
}

// tiny portrait renderer (renders each alien to a mini canvas for the HUD)
const IMPORTED_ALIEN_ASSETS={
  four_arms:{file:'fourarms.glb',height:4.25},
  heatblast:{file:'heatblast.glb',height:3.20},
  xlr8:{file:'xlr8_rigged.glb',height:2.80},
  echo_echo:{file:'echoecho.glb',height:2.55},
  cannonbolt:{file:'cannonbolt.glb',height:2.70}
};
const importedAlienCache=Object.create(null);
const importedAlienPromises=Object.create(null);
function importedMeshFilter(kind){
  if(kind!=='echo_echo') return null;
  return (n,ni,json)=>{ const m=json.nodes[ni].mesh; return m===55||m===56||m===57; };
}
async function getImportedAlien(kind){
  if(!IMPORTED_ALIEN_ASSETS[kind]||!window.MiniGLTF) return null;
  if(importedAlienCache[kind]) return importedAlienCache[kind];
  if(!importedAlienPromises[kind]){
    const spec=IMPORTED_ALIEN_ASSETS[kind];
    importedAlienPromises[kind]=MiniGLTF.load('assets/aliens/'+spec.file,{meshFilter:importedMeshFilter(kind)}).then(root=>{
      root.traverse(o=>{ if(o.isMesh){ o.castShadow=true; o.receiveShadow=true; o.frustumCulled=false; } });
      // Cannonbolt's supplied bind pose is authored on its side. Rotate the entire imported
      // rig before height fitting so its long shell axis becomes the game's Y/up axis.
      if(kind==='cannonbolt'){ root.rotation.z=Math.PI/2; root.userData.axisCorrection='Z+90'; }
      root.updateMatrixWorld(true);
      const box=new THREE.Box3().setFromObject(root), h=Math.max(.0001,box.max.y-box.min.y);
      const k=spec.height/h; root.scale.setScalar(k); root.updateMatrixWorld(true);
      const b2=new THREE.Box3().setFromObject(root); root.position.y-=b2.min.y; root.updateMatrixWorld(true);
      root.userData.importedKind=kind; root.userData.targetHeight=spec.height;
      importedAlienCache[kind]=root;
      return root;
    }).catch(err=>{ console.warn('[ALIEN GLB] '+kind+' failed:',err); return null; });
  }
  return importedAlienPromises[kind];
}
function _boneFind(bones,tests){
  for(const re of tests){ const b=bones.find(x=>re.test((x.name||'').toLowerCase())); if(b) return b; }
  return null;
}
function buildImportedBoneMap(root,kind){
  // MiniGLTF preserves glTF joint nodes as Object3D/Group instances rather than
  // THREE.Bone instances. Three's SkinnedMesh/Skeleton accepts these as bone objects
  // because it consumes their matrixWorld, so collect the actual skeleton.bones instead
  // of filtering on isBone (the previous filter silently produced an empty map).
  const bones=[]; const seen=new Set();
  root.traverse(o=>{
    if(!o.isSkinnedMesh || !o.skeleton) return;
    for(const b of (o.skeleton.bones||[])){ if(b && !seen.has(b)){ seen.add(b); bones.push(b); } }
  });
  const all=bones.map(b=>(b.name||'').toLowerCase());
  const map={kind,bones,base:new Map()};
  bones.forEach(b=>map.base.set(b,b.quaternion.clone()));
  const pick=(...tests)=>_boneFind(bones,tests);
  if(kind==='xlr8'){
    map.hips=pick(/^_?rootjoint/,/^spine_01/); map.chest=pick(/^spine\.001_02/,/^spine\.002_03/); map.head=pick(/^spine\.006_07/,/^head/);
    map.lUpper=pick(/upper_arm\.l/); map.lFore=pick(/forearm\.l/); map.rUpper=pick(/upper_arm\.r/); map.rFore=pick(/forearm\.r/);
    map.lThigh=pick(/thigh\.l/,/upper_leg\.l/); map.lShin=pick(/shin\.l/,/calf\.l/); map.rThigh=pick(/thigh\.r/,/upper_leg\.r/); map.rShin=pick(/shin\.r/,/calf\.r/);
  } else if(kind==='heatblast'){
    map.hips=pick(/bip_pelvis/); map.chest=pick(/bip_spine_2/,/bip_spine_1/); map.head=pick(/bip_head/);
    map.lUpper=pick(/bip_upperarm_l/); map.lFore=pick(/bip_lowerarm_l/); map.rUpper=pick(/bip_upperarm_r/); map.rFore=pick(/bip_lowerarm_r/);
    map.lThigh=pick(/bip_hip_l/,/bip_thigh_l/,/bip_upperleg_l/); map.lShin=pick(/bip_knee_l/,/bip_calf_l/,/bip_lowerleg_l/); map.rThigh=pick(/bip_hip_r/,/bip_thigh_r/,/bip_upperleg_r/); map.rShin=pick(/bip_knee_r/,/bip_calf_r/,/bip_lowerleg_r/);
  } else if(kind==='cannonbolt'){
    map.hips=pick(/^hips_01/); map.chest=pick(/^chest_07/,/^spine_06/); map.head=pick(/head/);
    map.lUpper=pick(/^left arm_029/); map.lFore=pick(/^left elbow_030/); map.rUpper=pick(/^right arm_09/); map.rFore=pick(/^right elbow_010/);
    map.lThigh=pick(/wideleg[_\.].*l_/,/wideleg_l/); map.lShin=pick(/wideknee[_\.].*l_/,/wideknee_l/); map.rThigh=pick(/wideleg[_\.].*r_/,/wideleg_r/); map.rShin=pick(/wideknee[_\.].*r_/,/wideknee_r/);
  } else if(kind==='echo_echo'){
    // Echo's exported rig uses numbered Blender bones. Its first 57 joints are stable across
    // the three Echo skins: root, spine/head chain, then the paired limbs. Use joint order
    // plus local X sign to identify the limbs rather than depending on generated suffixes.
    map.hips=bones[0]||null; map.chest=bones[2]||bones[1]||null; map.head=bones.find(b=>/bone\.0?43/i.test(b.name||''))||bones[1]||null;
    const limbs=bones.slice(1).filter(b=>{const n=(b.name||'').toLowerCase();return !n.includes('end');});
    const left=limbs.filter(b=>b.position.x>0.01), right=limbs.filter(b=>b.position.x<-0.01);
    const byY=(arr,hi)=>arr.slice().sort((a,b)=>b.position.y-a.position.y)[0]||null;
    map.lUpper=byY(left); map.rUpper=byY(right);
    const childOf=(b)=>b&&b.children.find(c=>bones.includes(c)&&!/_end/i.test(c.name||''));
    map.lFore=childOf(map.lUpper); map.rFore=childOf(map.rUpper);
    const legs=limbs.filter(b=>Math.abs(b.position.x)>0.01&&b.position.y<0); const ll=legs.filter(b=>b.position.x>0), rr=legs.filter(b=>b.position.x<0);
    map.lThigh=byY(ll); map.rThigh=byY(rr); map.lShin=childOf(map.lThigh); map.rShin=childOf(map.rThigh);
  }
  map.ready=!!(map.hips||map.chest||map.lUpper||map.rUpper);
  return map;
}
function cloneImportedAlienRig(source,kind){
  // Clone the hierarchy and explicitly map every source bone object to its clone.
  // Name-based remapping is unsafe for Echo Echo because several exported skins can
  // contain duplicate/generated bone names.
  const cloned=source.clone(true);
  const srcObjects=[], cloneObjects=[];
  source.traverse(o=>srcObjects.push(o));
  cloned.traverse(o=>cloneObjects.push(o));
  const objectMap=new Map();
  const n=Math.min(srcObjects.length,cloneObjects.length);
  for(let i=0;i<n;i++) objectMap.set(srcObjects[i],cloneObjects[i]);

  const srcSkins=[], cloneSkins=[];
  source.traverse(o=>{if(o.isSkinnedMesh) srcSkins.push(o);});
  cloned.traverse(o=>{if(o.isSkinnedMesh) cloneSkins.push(o);});

  cloneSkins.forEach((sm,i)=>{
    const src=srcSkins[i]; if(!src||!src.skeleton) return;
    const srcSkel=src.skeleton;
    const bones=(srcSkel.bones||[]).map((b,bi)=>objectMap.get(b)||null);
    const valid=bones.every(Boolean);
    if(!valid){ console.warn('[ALIEN GLB] '+kind+' skeleton remap incomplete on skin '+i); return; }
    const inverses=(srcSkel.boneInverses||[]).map(m=>m.clone());
    const skeleton=new THREE.Skeleton(bones,inverses);
    sm.bind(skeleton,src.bindMatrix.clone());
    sm.bindMode=src.bindMode||'attached';
    sm.frustumCulled=false;
    if(sm.normalizeSkinWeights) sm.normalizeSkinWeights();
    const mats=Array.isArray(sm.material)?sm.material:[sm.material];
    mats.forEach(mat=>{if(mat){mat.side=THREE.DoubleSide;mat.transparent=false;mat.opacity=1;mat.depthWrite=true;mat.needsUpdate=true;}});
    sm.userData.gltfSkinned=true;
  });
  cloned.traverse(o=>{if(o.isMesh){o.frustumCulled=false;o.castShadow=true;o.receiveShadow=true;}});
  cloned.updateMatrixWorld(true);
  return cloned;
}
function attachImportedAlienVisual(model,kind){
  if(!IMPORTED_ALIEN_ASSETS[kind]) return;
  getImportedAlien(kind).then(imported=>{
    if(!imported || !model || model.userData.kind !== kind) return;
    const old=model.userData.importedVisual; if(old) model.remove(old);
    let hasSkinned=false,meshCount=0; imported.traverse(o=>{ if(o.isSkinnedMesh) hasSkinned=true; if(o.isMesh) meshCount++; });
    // Four Arms' supplied GLB is intentionally preserved as an asset, but it has no skeleton.
    // Keep the game's live procedural rig active for that model instead of replacing it with a frozen mesh.
    if(!hasSkinned){
      console.warn('[ALIEN GLB] '+kind+' has no skin; keeping procedural articulated rig.');
      return;
    }
    const rig=cloneImportedAlienRig(imported,kind);
    let validSkin=true; rig.traverse(o=>{ if(o.isSkinnedMesh && (!o.geometry.getAttribute('skinIndex') || !o.geometry.getAttribute('skinWeight'))) validSkin=false; });
    if(!rig||!meshCount||!validSkin){ console.warn('[ALIEN GLB] '+kind+' produced an invalid skin/render state; keeping procedural fallback.'); return; }
    model.add(rig); model.userData.importedVisual=rig;
    const map=buildImportedBoneMap(rig,kind); rig.userData.bonePoseMap=map;
    const parts=model.userData.parts; if(parts){ parts.importedVisual=rig; parts.importedBoneMap=map; }
    rig.updateMatrixWorld(true);
    // Only retire the old body after the imported rig has been attached and rebound.
    const procedural=model.userData.proceduralRoot; if(procedural) procedural.visible=false;
    console.log('[ALIEN GLB] '+kind+' ready: '+meshCount+' meshes, '+map.bones.length+' bones, live skin binding.', rig.getWorldPosition(new THREE.Vector3()), rig.getWorldScale(new THREE.Vector3()));
  });
}
// Translate the game's mature procedural pose into the imported GLB skeleton. We reset each
// bone to its authored bind rotation first, then add the current game pose. This means the
// supplied anatomy stays intact while walk/run/jump/attack/hurt/death all drive real joints.
function poseImportedAlien(parts,dt,inp,P){
  const m=parts&&parts.importedBoneMap; if(!m||!m.ready) return;
  const d=dt>0?Math.min(1,dt*32):1;
  // Apply an additive local-space pose on top of the authored bind orientation.
  // This preserves the supplied mesh's anatomical axes while letting the shared game rig drive real bones.
  const smooth=(b,tx,ty=0,tz=0,rate=d)=>{
    if(!b) return;
    const base=m.base.get(b); if(!base) return;
    const delta=new THREE.Quaternion().setFromEuler(new THREE.Euler(tx,ty,tz,'XYZ'));
    const target=base.clone().multiply(delta);
    b.quaternion.slerp(target,Math.min(1,rate));
  };
  const A=parts.anim||{};
  const ph=A.ph||0;
  const combat=(inp.atk||inp.hurt||inp.dead)?0.35:1;
  const combatScale=combat;
  const kind=m.kind;
  const sign=kind==='cannonbolt'?-1:1;

  let hipsX=P.hipsRX, hipsY=P.hipsRY*.55, hipsZ=P.hipsRZ*.55;
  let chestX=P.chX*.9, chestY=P.chY*.55, chestZ=P.chZ*.8;
  let headX=P.hdX*.85, headY=P.hdY*.75, headZ=P.hdZ*.6;
  let lUpperX=P.sL.x*sign, lUpperY=P.sL.y, lUpperZ=P.sL.z;
  let rUpperX=P.sR.x*sign, rUpperY=P.sR.y, rUpperZ=P.sR.z;
  let lForeX=P.eL*sign, rForeX=P.eR*sign;
  let lThighX=P.tLx*sign, rThighX=P.tRx*sign;
  let lShinX=-P.kL*sign, rShinX=-P.kR*sign;

  // -----------------------------------------------------------------------
  // XLR8 calibration: convert the shared sprint cycle into the long-legged,
  // aerodynamic stride the supplied rig was built for. The extra phase term
  // makes the hips and knees alternate visibly instead of simply scaling the
  // same walk pose upward.
  // -----------------------------------------------------------------------
  if(kind==='xlr8'){
    const sprint=_ez((inp.superT||0)*1.15);
    const agile=_ez((sprint*.8)+(inp.moving?.32:0));
    const strideGain=1.18+.52*sprint;
    const kneeGain=1.10+.42*sprint;
    const gait=Math.sin(ph)*(.07+.14*sprint);
    lThighX=Math.max(-2.15,Math.min(2.15,lThighX*strideGain + gait*combatScale));
    rThighX=Math.max(-2.15,Math.min(2.15,rThighX*strideGain - gait*combatScale));
    lShinX=Math.max(-2.3,Math.min(2.3,lShinX*kneeGain + Math.max(0,Math.sin(ph))*0.16*sprint*combatScale));
